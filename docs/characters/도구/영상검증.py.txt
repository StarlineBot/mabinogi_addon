# 기준 = 그 캐릭터 서 있기 영상 0번 장. 동작 영상(초록 배경 GIF)마다 장별로:
#  1) 머리 크기: 서 있기 머리(위 35% — 후광·뿔·얼굴)를 템플릿으로, 배율을 바꿔가며 가장 잘 맞는 배율 = 그 장의 머리 크기(서 있기=1.00)
#     눈 감기·표정과 무관(머리 윤곽 전체로 맞춤).
#  2) 크기 흔들림: 한 영상 안 머리 크기 최대/최소 — 6% 넘으면 문제(장이 넘어갈 때 커졌다 작아짐)
#  3) 몸 비율: '서 있는 장'(머리 크기로 맞춘 뒤 서 있기와 실루엣 겹침 ≥0.80)의 키 ÷ 서 있기 키 — ±5% 넘으면 문제(뚱뚱/길쭉)
#  4) 시작·끝 자세: 머리 크기로 맞춘 첫 장·끝 장 ↔ 서 있기 실루엣 겹침(서기에서 시작/끝나는 동작만) — 0.85 미만이면 문제
#  5) 색: 어두운·중간·밝은 띠 Lab 평균 ΔE ≤6
#  6) 화면 끝 잘림: 몸이 영상 위·양옆 끝에 닿은 장(경고)
# 결과: 동작마다 '통과' 또는 '문제: …' 한 줄 + 숫자. 사람이 그림을 안 봐도 되게.
import sys, glob, os, json
import numpy as np, cv2
from PIL import Image

def key(img):
    f = img.astype(np.float32); b, g, r = f[..., 0], f[..., 1], f[..., 2]
    al = np.clip(1 - ((g - np.maximum(r, b)) - 25) / 70, 0, 1)
    m = (al > 0.5).astype(np.uint8); n, l, st, _ = cv2.connectedComponentsWithStats(m)
    if n <= 1: return np.dstack([img, np.zeros(img.shape[:2], np.uint8)])
    bi = 1 + int(np.argmax(st[1:, 4]))
    # 11-28: Gemini GIF 오른쪽 아래 별 표시(워터마크)가 2%를 넘어 상자를 키움(재현: 레나 GIF 시작 겹침 0.34) →
    # 큰 덩어리 상자를 25% 넓힌 범위와 겹치는 조각만 남김(몸 옆 마법 효과는 남고, 구석 표시는 빠짐)
    x, y, w, h = st[bi, :4]; px, py = int(w * 0.25), int(h * 0.25)
    near = lambda i: not (st[i, 0] > x + w + px or st[i, 0] + st[i, 2] < x - px or st[i, 1] > y + h + py or st[i, 1] + st[i, 3] < y - py)
    keep = np.isin(l, [i for i in range(1, n) if st[i, 4] >= st[bi, 4] * 0.02 and near(i)])
    al = al * cv2.dilate(keep.astype(np.uint8), np.ones((3, 3), np.uint8))
    g2 = np.minimum(g, np.maximum(r, b) + 8)
    return np.dstack([b, g2, r, al * 255]).clip(0, 255).astype(np.uint8)

def load(p):
    """GIF·mp4·png 모두 — 초록 배경을 빼서 장 목록으로."""
    p = str(p); out = []
    if p.lower().endswith((".mp4", ".mov", ".webm")):
        cap = cv2.VideoCapture(p)
        while True:
            ok, f = cap.read()
            if not ok: break
            out.append(key(f))
        return out
    g = Image.open(p)
    for i in range(getattr(g, "n_frames", 1)):
        g.seek(i); out.append(key(cv2.cvtColor(np.asarray(g.convert("RGB")), cv2.COLOR_RGB2BGR)))
    return out

def tight(a):
    ys, xs = np.where(a[..., 3] > 128); return a[ys.min():ys.max() + 1, xs.min():xs.max() + 1]

def gray(a):  # 회색 바탕에 합성한 밝기(템플릿 맞춤용)
    al = a[..., 3:] / 255.0
    return (cv2.cvtColor(a[..., :3], cv2.COLOR_BGR2GRAY)[..., None] * al + 128 * (1 - al))[..., 0].astype(np.float32)

SCALES = np.linspace(0.80, 1.25, 46)

def _rot(img, ang):
    if ang == 0: return img
    h, w = img.shape[:2]; M = cv2.getRotationMatrix2D((w / 2, h / 2), ang, 1.0)
    return cv2.warpAffine(img, M, (w, h), borderValue=(0, 0, 0, 0))

def head_scale(ref_head, frame):
    """frame 머리 크기(ref=1)와 맞춤 신뢰도. 고개 기울기(±24°)도 찾음 — 기울어진 장에서 크기를 잘못 재지 않게."""
    rg = gray(ref_head); best = (-2, 1.0)
    for ang in (0, -8, 8, -16, 16, -24, 24):
        rr = gray(_rot(ref_head, ang)) if ang else rg
        for s in SCALES:
            f = cv2.resize(frame, None, fx=1 / s, fy=1 / s, interpolation=cv2.INTER_AREA)
            if f.shape[0] < rr.shape[0] or f.shape[1] < rr.shape[1]:
                f = np.pad(f, ((0, max(0, rr.shape[0] - f.shape[0] + 2)), (0, max(0, rr.shape[1] - f.shape[1] + 2)), (0, 0)))
            r = cv2.matchTemplate(gray(f), rr, cv2.TM_CCOEFF_NORMED).max()
            if r > best[0]: best = (r, s)
        if ang == 0 and best[0] > 0.85: break      # 똑바른 장은 빨리 끝
    return best[1], best[0]

def place(m, h, w):
    o = np.zeros((h, w), bool); mh, mw = m.shape; x = (w - mw) // 2; o[h - mh:, x:x + mw] = m; return o

def iou(a, b):
    h, w = max(a.shape[0], b.shape[0]), max(a.shape[1], b.shape[1]) + 2
    pa, pb = place(a, h, w), place(b, h, w); return float((pa & pb).sum() / max(1, (pa | pb).sum()))

def bands(fs):
    px = np.concatenate([cv2.cvtColor(a[..., :3], cv2.COLOR_BGR2LAB).astype(float)[a[..., 3] > 200] for a in fs]); L = px[:, 0]
    return np.array([px[(L >= lo) & (L < hi)].mean(0) for lo, hi in [(0, 90), (90, 190), (190, 256)]])

def check(stand_gif, motion_gif, from_stand=True, to_stand=True):
    S = tight(load(stand_gif)[0]); SH = S.shape[0]; ref_head = S[: int(SH * 0.35)]; sm = S[..., 3] > 128
    F = load(motion_gif); res = {"장": len(F)}
    hs, conf, src = [], [], []
    area0 = float((S[..., 3] > 128).sum())
    for a in F:
        t = tight(a); s, c = head_scale(ref_head, t)
        src.append("머리" if c > 0.75 else "못 잼")   # 머리가 기울거나 가려져 안 맞는 장은 크기 판정에서 뺌(몸 넓이는 자세 따라 변해 못 씀 — Starline 시험 11-27)
        hs.append(s); conf.append(c)
    hs = np.array(hs); good = np.array([x == "머리" for x in src])
    res["판정 못한 장"] = [i for i, x in enumerate(src) if x != "머리"]
    med = float(np.median(hs[good])) if good.any() else float(np.median(hs))
    res["머리크기"] = round(med, 3); res["머리 흔들림%"] = round((hs[good].max() / hs[good].min() - 1) * 100, 1) if good.any() else None
    norm = [cv2.resize(tight(a), None, fx=1 / s, fy=1 / s, interpolation=cv2.INTER_AREA)[..., 3] > 128 for a, s in zip(F, hs)]
    ious = [iou(sm, m) for m in norm]
    standing = [i for i, v in enumerate(ious) if v >= 0.85 and good[i]]
    if standing:
        r = [norm[i].shape[0] / SH for i in standing]; res["서 있는 장 키비"] = (round(min(r), 3), round(max(r), 3))
    res["시작 겹침"] = round(ious[0], 2); res["끝 겹침"] = round(ious[-1], 2)
    res["색 ΔE"] = [round(x, 1) for x in np.linalg.norm(bands(F[::3]) - bands([S]), axis=1)]
    edge = [i for i, a in enumerate(F) if (a[:2, :, 3] > 128).any() or (a[:, :2, 3] > 128).any() or (a[:, -2:, 3] > 128).any()]
    res["화면끝 닿은 장"] = edge
    prob = []
    if res["머리 흔들림%"] is not None and res["머리 흔들림%"] > 6: prob.append(f"크기 흔들림 {res['머리 흔들림%']}%(장 {int(np.argmin(hs))}↔{int(np.argmax(hs))})")
    if standing and (min(r) < 0.95 or max(r) > 1.05): prob.append(f"몸 비율 다름(서 있는 장 키 {min(r):.2f}~{max(r):.2f}배)")
    if from_stand and ious[0] < 0.85: prob.append(f"시작 자세가 서 있기와 다름({ious[0]:.2f})")
    if to_stand and ious[-1] < 0.85: prob.append(f"끝 자세가 서 있기와 다름({ious[-1]:.2f})")
    if max(res["색 ΔE"]) > 6: prob.append(f"색 다름 ΔE {max(res['색 ΔE'])}")
    res["판정"] = "통과" if not prob else "문제: " + " / ".join(prob)
    if edge: res["판정"] += f" (경고: 화면 끝 닿음 {len(edge)}장)"
    res["장별 머리크기"] = [round(x, 2) for x in hs]
    return res

if __name__ == "__main__" and sys.argv[1:2] != ["폴더"]:
    # 사용: python 영상검증.py <캐릭터 영상 폴더> [동작파일 …]  — 기본은 폴더의 *.gif 전부(0_서있기 제외)
    # 서 있기에서 시작/끝나지 않는 동작은 이름표로: NOSTART(앞 영상 장에서 시작), NOEND(앉은 채·매달린 채 끝)
    V = sys.argv[1].rstrip("/") + "/"
    spec = json.loads(sys.argv[2]) if len(sys.argv) > 2 else {}
    for p in sorted(glob.glob(V + "*.gif"), key=lambda x: (len(os.path.basename(x)), x)):
        n = os.path.basename(p)[:-4]
        if n.startswith("0_") or "1차" in n: continue
        fs, ts = spec.get(n, [True, True])
        r = check(V + "0_서있기.gif", p, fs, ts)
        print(f"{n:16s} {r['판정']}")
        print(f"{'':16s} 머리 크기 못 잰 장 {r['판정 못한 장']}"); print(f"{'':16s} 머리 {r['머리크기']}(흔들림 {r['머리 흔들림%']}%) 서있는장 키비 {r.get('서 있는 장 키비')} 시작/끝 겹침 {r['시작 겹침']}/{r['끝 겹침']} 색ΔE {r['색 ΔE']}")


def check_folder(folder, stand_gif, loop=None):
    """요청 폴더 하나(1_시작.png·2_끝.png·결과.*) 검증 → 요청.md '검증' 칸에 적음.
    11-28 고침(재현: Starline·다래 결과로 오판 확인): 색은 같은 자세끼리(첫 장↔시작 그림, 끝 장↔끝 그림)만 비교 — 영상 전체 평균은 자세가 바뀌면
    색 비율이 달라져 틀림. 크기는 머리 맞춤 신뢰 0.85 넘는 장만. 시작·끝 그림이 같으면(반복 동작) 끝↔처음 이음매로 판정."""
    folder = folder.rstrip("/") + "/"
    res_file = ([f for f in glob.glob(folder + "결과.*")] or [None])[0]
    if res_file is None: return "결과 파일 없음"
    F = load(res_file)
    S = tight(load(stand_gif)[0]); SH = S.shape[0]; ref_head = S[: int(SH * 0.35)]
    s_img, e_img = load(folder + "1_시작.png")[0], load(folder + "2_끝.png")[0]
    if loop is None: loop = bool(np.array_equal(s_img, e_img))
    # 끝 칸 비움 작업(끌기·구석 앉기 등 — 요청.md에 '끝 칸 비움'): 2_끝.png는 결과 마지막 장이라 끝 겹침은 보지 않음
    md0 = folder + "요청.md"
    hold = os.path.exists(md0) and "끝 칸 비움" in open(md0, encoding="utf-8").read().split("## 3. 검증")[0]
    # 11-28: Gemini 앱 GIF는 640×360, 시작 그림은 1280×720 — 머리 맞춤 범위(0.80~1.25)를 벗어나 겹침 0.34로 오판(재현: 레나 L9).
    # 카메라 고정이라 첫 장 키를 시작 그림 키에 맞춘 배율 하나로 모든 장을 먼저 맞춤.
    f0 = tight(s_img).shape[0] / tight(F[0]).shape[0]
    if abs(f0 - 1) > 0.15:
        F = [cv2.resize(a, None, fx=f0, fy=f0, interpolation=cv2.INTER_LINEAR) for a in F]
    def norm(a):
        t = tight(a); sc, c = head_scale(ref_head, t)
        return cv2.resize(t, None, fx=1 / sc, fy=1 / sc, interpolation=cv2.INTER_AREA)[..., 3] > 128, sc, c
    st0 = norm(s_img)[0]; en0 = norm(e_img)[0]
    a0 = norm(F[0])[0]; a1 = norm(F[-1])[0]
    i_start, i_end = iou(st0, a0), iou(en0, a1)
    hs = []
    for a in F:
        sc, c = head_scale(ref_head, tight(a))
        if c > 0.85: hs.append(sc)
    spread = (max(hs) / min(hs) - 1) * 100 if len(hs) >= 2 else 0.0
    def bands1(a):
        px = cv2.cvtColor(a[..., :3], cv2.COLOR_BGR2LAB).astype(float)[a[..., 3] > 200]; L = px[:, 0]
        return np.array([px[(L >= lo) & (L < hi)].mean(0) if ((L >= lo) & (L < hi)).sum() > 30 else np.zeros(3) for lo, hi in [(0, 90), (90, 190), (190, 256)]])
    c_start = np.linalg.norm(bands1(F[0]) - bands1(s_img), axis=1); c_end = np.linalg.norm(bands1(F[-1]) - bands1(e_img), axis=1)
    g = [cv2.cvtColor(a[..., :3], cv2.COLOR_BGR2GRAY).astype(float) for a in F]; m = [a[..., 3] > 128 for a in F]
    d = lambda i, j: float(np.abs(g[i] - g[j])[m[i] | m[j]].mean())
    nb = [d(i, i + 1) for i in range(len(F) - 1)]; seam = d(len(F) - 1, 0)
    edge = [i for i, a in enumerate(F) if (a[:2, :, 3] > 128).any() or (a[:, :2, 3] > 128).any() or (a[:, -2:, 3] > 128).any()]
    prob = []
    if i_start < 0.85: prob.append(f"시작 장면이 1_시작.png와 다름(겹침 {i_start:.2f})")
    if loop:
        if seam > max(nb) * 1.2: prob.append(f"반복 이음매가 튐(끝→처음 차 {seam:.1f}, 이웃 장 최대 {max(nb):.1f})")
    elif i_end < 0.88 and not hold: prob.append(f"끝 장면이 2_끝.png와 다름(겹침 {i_end:.2f})")
    if spread > 6: prob.append(f"크기 흔들림 {spread:.1f}%")
    if max(c_start[1:].max(), c_end[1:].max()) > 6 or max(c_start[0], c_end[0]) > 8:
        prob.append(f"색 다름(시작 ΔE {[round(float(x),1) for x in c_start]}, 끝 ΔE {[round(float(x),1) for x in c_end]})")
    verdict = "통과" if not prob else "다시: " + " / ".join(prob)
    lines = [f"- 판정: **{verdict}**",
             f"- 시작 겹침 {i_start:.2f}" + (f" · 반복 이음매 끝→처음 차 {seam:.1f}(이웃 장 평균 {np.mean(nb):.1f}, 최대 {max(nb):.1f})" if loop else (" · 끝 칸 비움 작업 — 끝 겹침 안 봄" if hold else f" · 끝 겹침 {i_end:.2f}(기준 0.88)")),
             f"- 머리 크기 흔들림 {spread:.1f}%(기준 6%, 머리 맞춤 잘 된 {len(hs)}장으로) · 색 ΔE 시작 {[round(float(x),1) for x in c_start]} 끝 {[round(float(x),1) for x in c_end]}(어두운·중간·밝은, 기준 8·6·6)",
             f"- 화면 끝 닿은 장 {len(edge)}개 · 장 수 {len(F)}"]
    md = folder + "요청.md"
    t = open(md, encoding="utf-8").read() if os.path.exists(md) else ""
    head = t.split("## 3. 검증")[0]
    open(md, "w", encoding="utf-8").write(head + "## 3. 검증 (기획이 씀)\n" + "\n".join(lines) + "\n")
    return verdict


if __name__ == "__main__" and sys.argv[1:2] == ["폴더"]:
    # 사용: python 영상검증.py 폴더 <요청 폴더> <서 있기 gif>
    print(check_folder(sys.argv[2], sys.argv[3]))

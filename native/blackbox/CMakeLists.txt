cmake_minimum_required(VERSION 3.24)
project(mabinogi_blackbox LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows SDK and MSVC are required; no non-Windows capture fallback")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(mabinogi_blackbox SHARED capture.cpp audio.cpp)
target_compile_features(mabinogi_blackbox PRIVATE cxx_std_20)
target_compile_definitions(mabinogi_blackbox PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mabinogi_blackbox PRIVATE d3d11 dxgi windowsapp runtimeobject mmdevapi ole32)
add_executable(blackbox_codec_host codec_host.cpp)
target_compile_features(blackbox_codec_host PRIVATE cxx_std_20)
target_compile_definitions(blackbox_codec_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
